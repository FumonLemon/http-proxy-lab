# rules.txt —— 代理的名单配置。存盘即生效：不用重启代理，也不用重新编译。
#
# 语法：# 开头是注释；[小节名] 切换名单；其余每行一个条目；空行忽略。
#       主机名匹配不区分大小写；只比主机名，不含端口、不含路径。
#       文件读不到（或为空）时，代理回落用内置默认名单：127.0.0.1 / 192.0.2.1 / guide.test。

[site]
# 网站过滤：打到名单里主机的请求，代理直接回 403，不去连原服务器。
# 只对 http:// 的绝对形式请求有意义（https 走 CONNECT，在解析阶段就回 501）。
# 演示用 example.com：它既能演 403，又不会把缓存靶子（本机 127.0.0.1 / 各镜像站）挡掉。
example.com


[user]
# 用户过滤：来源 IP 在名单里就不服务（回 403）。
# 单机上真实来源恒为 127.0.0.1，直接把它写进来会把自己封掉。演示"两个用户"的办法：
#   回环段 127.0.0.0/8 整段都是本机，127.0.0.2 是合法的第二个源地址，且不影响浏览器走 127.0.0.1。
#   名单里封 127.0.0.2，然后用 curl 绑定这个源地址来扮演"被封的用户"：
#     curl --interface 127.0.0.2 -x http://127.0.0.1:8080 http://info.cern.ch/   → 403 用户过滤
#     curl                      -x http://127.0.0.1:8080 http://info.cern.ch/   → 200 正常
# 192.0.2.1 是文档专用地址（TEST-NET-1），永远不会是真实来源，也可以留作"别人"的占位。
127.0.0.2

[redirect]
# 网站引导：一行两个字段（域名 + 目标），空白分隔。目标有两种写法：
#   1) self:文件路径 → 代理把这个文件当页面直接回给你，地址栏仍保持原域名（钓鱼/引导演示用这个）
#   2) http://…     → 代理回 302 跳过去，但浏览器地址栏会变成那个网址
guide.test    self:fake.html

[auth]
# 代理认证（按账号的"用户过滤"）：一行一个用户，格式
#     用户名:密码                  该用户不受限
#     用户名:密码:deny=a.com,b.com 该用户被禁止访问这些主机
# 注意：只要这一段里有任何一行生效，所有请求都必须带凭据（否则 407）。
#       想临时关掉认证，就把下面两行的 # 加回去 —— 存盘即生效，不用重启。
#alice:1234
#bob:1234:deny=example.com,guide.test
